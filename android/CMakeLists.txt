# ─────────────────────────────────────────────────────────────────────────────
# OrbitLooper — Android APK
#
# Built via android/app/build.gradle + Gradle wrapper.
# Do NOT run cmake directly — Gradle drives the CMake invocation.
#
# PREREQUISITE: Run the Windows desktop configure first so that JUCE is
# fetched and juceaide.exe is built:
#
#   cmake -S . -B build -G "Visual Studio 17 2022" -A x64
#   cmake --build build --config Release
#
# Then build the APK from this directory:
#   .\gradlew.bat assembleRelease_Release
# ─────────────────────────────────────────────────────────────────────────────

cmake_minimum_required(VERSION 3.22)
project(OrbitLooperAndroid VERSION 1.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ── Locate JUCE fetched by the desktop build ──────────────────────────────────
# The main CMakeLists.txt uses FetchContent; JUCE lands in build/_deps/juce-src.
# The path below is relative to android/ (one level up → OrbitLooper root).
set(JUCE_SRC_DIR  "${CMAKE_CURRENT_SOURCE_DIR}/../build/_deps/juce-src")
set(JUCE_BUILD_TOOLS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/_deps/juce-build/tools")

if(NOT EXISTS "${JUCE_SRC_DIR}")
    message(FATAL_ERROR
        "JUCE source not found at: ${JUCE_SRC_DIR}\n"
        "Run the Windows desktop configure first to fetch JUCE:\n"
        "  cmake -S . -B build -G \"Visual Studio 17 2022\" -A x64")
endif()

# ── juceaide pre-bootstrap (Android cross-compilation) ───────────────────────
# When Gradle cross-compiles for Android, JUCE's juceaide/CMakeLists.txt tries
# to re-invoke cmake to build juceaide for the host — which fails because
# rc.exe is not in the Gradle process PATH.
#
# Fix: pre-create JUCEToolsExport.cmake pointing at the juceaide.exe already
# built by the desktop Windows build, then apply the patched juceaide
# CMakeLists.txt (Source/patches/juce_juceaide_android.patched.cmake) which
# short-circuits when that file is already present.
if(CMAKE_CROSSCOMPILING)
    # Find juceaide built by the desktop build (platform-agnostic)
    if(CMAKE_HOST_WIN32)
        set(_juceaide_name "juceaide.exe")
    else()
        set(_juceaide_name "juceaide")
    endif()
    set(_juceaide_exe
        "${JUCE_BUILD_TOOLS_DIR}/extras/Build/juceaide/juceaide_artefacts/Custom/${_juceaide_name}")
    get_filename_component(_juceaide_exe "${_juceaide_exe}" ABSOLUTE)

    if(NOT EXISTS "${_juceaide_exe}")
        message(FATAL_ERROR
            "juceaide not found at:\n  ${_juceaide_exe}\n"
            "Build the desktop Release first:\n"
            "  cmake -S . -B build -DCMAKE_BUILD_TYPE=Release\n"
            "  cmake --build build --target juceaide")
    endif()

    # Pre-create JUCEToolsExport.cmake in the location JUCE expects.
    # The binary dir for JUCE will be CMAKE_BINARY_DIR/juce_build (see add_subdirectory below).
    set(_juce_tools_dir "${CMAKE_BINARY_DIR}/juce_build/tools")
    if(NOT EXISTS "${_juce_tools_dir}/JUCEToolsExport.cmake")
        file(MAKE_DIRECTORY "${_juce_tools_dir}")
        file(WRITE "${_juce_tools_dir}/JUCEToolsExport.cmake"
            "# Auto-generated by android/CMakeLists.txt for Android cross-compilation.\n"
            "if(NOT TARGET juce_tools::juceaide)\n"
            "    add_executable(juce_tools::juceaide IMPORTED GLOBAL)\n"
            "    set_property(TARGET juce_tools::juceaide APPEND PROPERTY IMPORTED_CONFIGURATIONS CUSTOM)\n"
            "    set_target_properties(juce_tools::juceaide PROPERTIES\n"
            "        IMPORTED_LOCATION_CUSTOM \"${_juceaide_exe}\")\n"
            "endif()\n")
        message(STATUS "OrbitLooper: Pre-created JUCEToolsExport.cmake → ${_juceaide_exe}")
    endif()
endif()

# ── Patch juceaide/CMakeLists.txt for cross-compilation skip ─────────────────
# Apply the patched version before add_subdirectory so JUCE picks it up.
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/../Source/patches/juce_juceaide_android.patched.cmake"
    "${JUCE_SRC_DIR}/extras/Build/juceaide/CMakeLists.txt"
    COPYONLY
)

# ── JUCE ─────────────────────────────────────────────────────────────────────
# Re-use the already-fetched JUCE source; write build artifacts to juce_build/.
add_subdirectory("${JUCE_SRC_DIR}" juce_build)

# ── Android: rename .so → juce_jni + fix symbol visibility ───────────────────
# JUCE's Java bootstrap (JuceApp.java) calls System.loadLibrary("juce_jni"),
# so the native library MUST be named libjuce_jni.so inside the APK.
#
# Also: JUCE's CMake sets C/CXX_VISIBILITY_PRESET=hidden on all targets, which
# hides juce_CreateApplication from the dynamic symbol table.  The JVM finds
# JNI_OnLoad via JNIEXPORT, but JUCE's juce_juceEventsAndroidStartApp() looks
# up juce_CreateApplication via dlsym() — it must be visible.  Setting
# CXX_VISIBILITY_PRESET=default on the Standalone target (which is a final
# application .so, not a reusable library) exports both symbols correctly.
if(CMAKE_SYSTEM_NAME STREQUAL "Android")
    set_target_properties(OrbitLooper_Standalone PROPERTIES
        OUTPUT_NAME          "juce_jni"
        C_VISIBILITY_PRESET  default
        CXX_VISIBILITY_PRESET default)

    target_link_options(OrbitLooper_Standalone PRIVATE
        "-Wl,-u,JNI_OnLoad"
    )
endif()
